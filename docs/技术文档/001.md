#### 1 录制并播放一段音频
        Rust跨平台音频处理:CPAL库的使用与特性
        use cpal::traits::{DeviceTrait, HostTrait, StreamTrait};
        use std::time::Duration;
        
        fn main() {
        let host = cpal::default_host();
        
            // 获取默认输入和输出设备
            let input_device = host.default_input_device().expect("No input device found");
            let output_device = host.default_output_device().expect("No output device found");
        
            // 获取默认输入配置
            let input_config = input_device.default_input_config().unwrap();
            let output_config = output_device.default_output_config().unwrap();
        
            // 创建共享缓冲区
            let buffer: Arc<Mutex<Vec<f32>>> = Arc::new(Mutex::new(Vec::with_capacity(960)));
        
            // 构建输入流（录音）
            let in_buffer = buffer.clone();
            let in_stream = input_device.build_input_stream(
                &input_config.into(),
                move |data: &[f32], _: &cpal::InputCallbackInfo| {
                    let mut lock = in_buffer.lock().unwrap();
                    lock.extend_from_slice(data);
                },
                |err| eprintln!("Input error: {}", err),
                None,
            ).unwrap();
        
            // 构建输出流（播放）
            let out_buffer = buffer;
            let out_stream = output_device.build_output_stream(
                &output_config.into(),
                move |data: &mut [f32], _: &cpal::OutputCallbackInfo| {
                    let mut lock = out_buffer.lock().unwrap();
                    if lock.len() >= data.len() {
                        data.copy_from_slice(&lock[..data.len()]);
                        lock.drain(..data.len());
                    }
                },
                |err| eprintln!("Output error: {}", err),
                None,
            ).unwrap();
        
            // 启动音频流
            in_stream.play().unwrap();
            out_stream.play().unwrap();
        
            // 播放 5 秒后停止
            std::thread::sleep(Duration::from_secs(5));
        }
